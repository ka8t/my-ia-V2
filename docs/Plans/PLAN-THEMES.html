// UI-FRONT/js/modules/settings.js - EXISTANT

export function toggleTheme() {
    window.CONFIG.theme = window.CONFIG.theme === 'light' ? 'dark' : 'light';
    document.documentElement.setAttribute('data-theme', window.CONFIG.theme);

    // Sauvegarde préférence utilisateur via API
    ApiService.updatePreferences({ theme: window.CONFIG.theme });
}

export function applyPreferences(prefs) {
    if (prefs.theme) {
        window.CONFIG.theme = prefs.theme;
        document.documentElement.setAttribute('data-theme', window.CONFIG.theme);
    }
}

6.3 Module JS

// UI-BACK/js/modules/system/appearance.js

export const SystemAppearance = {
    selectedTheme: null,

    async load() {
        const config = await adminApi.config.get('theme.global');
        this.selectedTheme = config || 'default';
        this.render();
    },

    render() {
        const themes = [
            { id: 'default', name: 'Default', accent: '#10a37f' },
            { id: 'corporate', name: 'Corporate', accent: '#0066cc' },
            { id: 'nature', name: 'Nature', accent: '#059669' },
            { id: 'sunset', name: 'Sunset', accent: '#ea580c' },
            { id: 'royal', name: 'Royal', accent: '#7c3aed' },
        ];

        const html = Templates.render('system-theme-section', {
            themes: themes.map(t => ({
                ...t,
                isActive: t.id === this.selectedTheme
            }))
        });

        document.getElementById('systemContent').innerHTML = html;
        this.bindEvents();
    },

    bindEvents() {
        document.querySelectorAll('.theme-card').forEach(card => {
            card.onclick = () => {
                this.selectedTheme = card.dataset.theme;
                this.render();
            };
        });

        document.querySelector('[data-action="save-theme"]').onclick = () => {
            this.save();
        };
    },

    async save() {
        await adminApi.config.set('theme.global', this.selectedTheme);
        showToast(t('admin_theme_saved'));
    }
};

7.3 Extension du code existant

Le code existant dans UI-FRONT/js/modules/settings.js gère déjà light/dark. Il faut l'étendre pour supporter le thème global admin :

// UI-FRONT/js/modules/settings.js - À ÉTENDRE

class ThemeService {
    constructor() {
        this.globalTheme = 'default';  // Défini par l'admin
        this.userChoice = null;        // 'global', 'light', 'dark'
        this.isDarkMode = false;
    }

    async init() {
        // 1. Charger le thème global depuis l'API
        try {
            const config = await fetch('/config/public').then(r => r.json());
            this.globalTheme = config.theme?.global || 'default';
        } catch (e) {
            this.globalTheme = 'default';
        }

        // 2. Charger le choix utilisateur depuis localStorage
        this.userChoice = localStorage.getItem('theme-choice') || 'global';
        this.isDarkMode = localStorage.getItem('theme-dark') === 'true';

        // 3. Appliquer
        this.apply();
    }

    apply() {
        let themeName;

        switch (this.userChoice) {
            case 'light':
                themeName = 'default';
                break;
            case 'dark':
                themeName = 'dark';
                break;
            case 'global':
            default:
                // Utiliser le thème admin + mode sombre si activé
                themeName = this.isDarkMode
                    ? `${this.globalTheme}-dark`
                    : this.globalTheme;
                break;
        }

        document.documentElement.setAttribute('data-theme', themeName);
        window.dispatchEvent(new CustomEvent('theme-changed', {
            detail: { theme: themeName }
        }));
    }

    setUserChoice(choice) {
        // choice: 'global', 'light', 'dark'
        this.userChoice = choice;
        localStorage.setItem('theme-choice', choice);
        this.apply();
    }

    toggleDarkMode() {
        this.isDarkMode = !this.isDarkMode;
        localStorage.setItem('theme-dark', this.isDarkMode);
        this.apply();
    }

    getCurrentTheme() {
        return document.documentElement.getAttribute('data-theme');
    }
}

export const Theme = new ThemeService();
