3.2 Comportement

  • Niveau 1 : 4 boutons de groupe (Application, Infrastructure, Sécurité, Communication)
  • Niveau 2 : Sous-onglets du groupe sélectionné
  • Persistance : Mémoriser le dernier groupe/onglet visité
  • Défaut : Application → Modes

4.3 Modification SystemModule.js

const SystemModule = {
    currentGroup: 'app',
    currentSubTab: 'modes',

    // Configuration des groupes
    groups: {
        app: {
            label: 'admin_system_group_app',
            icon: 'settings',
            tabs: ['modes', 'chat', 'speech', 'geo'],
            defaultTab: 'modes'
        },
        infra: {
            label: 'admin_system_group_infra',
            icon: 'server',
            tabs: ['llm', 'storage', 'indexation', 'appearance'],
            defaultTab: 'llm'
        },
        security: {
            label: 'admin_system_group_security',
            icon: 'shield',
            tabs: ['password', 'logging', 'debug'],
            defaultTab: 'password'
        },
        comm: {
            label: 'admin_system_group_comm',
            icon: 'mail',
            tabs: ['smtp', 'notifications'],
            defaultTab: 'smtp'
        }
    },

    async render(container) {
        // Restaurer état précédent
        this.currentGroup = localStorage.getItem('admin-system-group') || 'app';
        this.currentSubTab = localStorage.getItem('admin-system-subtab') || 'modes';

        // Vérifier que le subtab appartient au groupe
        if (!this.groups[this.currentGroup].tabs.includes(this.currentSubTab)) {
            this.currentSubTab = this.groups[this.currentGroup].defaultTab;
        }

        // Render layout avec groupes
        container.innerHTML = Templates.render('system-main-layout', {
            title: t('admin_system_title'),
            groups: Object.entries(this.groups).map(([id, cfg]) => ({
                id,
                label: t(cfg.label),
                icon: Icons[cfg.icon](),
                isActive: id === this.currentGroup
            }))
        });

        this.initGroupTabs();
        this.renderSubTabs();
        await this.loadCurrentSubTab();
    },

    initGroupTabs() {
        document.querySelectorAll('.system-group-btn').forEach(btn => {
            btn.onclick = () => this.switchGroup(btn.dataset.group);
        });
    },

    switchGroup(groupId) {
        this.currentGroup = groupId;
        this.currentSubTab = this.groups[groupId].defaultTab;

        localStorage.setItem('admin-system-group', groupId);
        localStorage.setItem('admin-system-subtab', this.currentSubTab);

        // Update UI
        document.querySelectorAll('.system-group-btn').forEach(btn => {
            btn.classList.toggle('active', btn.dataset.group === groupId);
        });

        this.renderSubTabs();
        this.loadCurrentSubTab();
    },

    renderSubTabs() {
        const group = this.groups[this.currentGroup];
        const html = group.tabs.map(tabId => `
            <button class="sub-tab ${tabId === this.currentSubTab ? 'active' : ''}"
                    data-subtab="${tabId}">
                ${t('admin_system_' + tabId)}
            </button>
        `).join('');

        document.getElementById('subTabsBar').innerHTML = html;

        // Bind events
        document.querySelectorAll('.sub-tab').forEach(tab => {
            tab.onclick = () => this.switchSubTab(tab.dataset.subtab);
        });
    },

    switchSubTab(subtabId) {
        this.currentSubTab = subtabId;
        localStorage.setItem('admin-system-subtab', subtabId);

        document.querySelectorAll('.sub-tab').forEach(t => {
            t.classList.toggle('active', t.dataset.subtab === subtabId);
        });

        this.loadCurrentSubTab();
    },

    async loadCurrentSubTab() {
        // Déléguer au sous-module approprié
        switch (this.currentSubTab) {
            case 'modes': await SystemModes.load(); break;
            case 'chat': await SystemChat.load(); break;
            case 'speech': await SystemSpeech.load(); break;
            case 'geo': await SystemGeo.load(); break;
            case 'llm': await SystemLLM.load(); break;
            case 'storage': await SystemStorage.load(); break;
            case 'indexation': await SystemIndexation.load(); break;
            case 'appearance': await SystemAppearance.load(); break;
            case 'password': await SystemPassword.load(); break;
            case 'logging': await SystemLogging.load(); break;
            case 'debug': await SystemDebug.load(); break;
            case 'smtp': await SystemSmtp.load(); break;
            case 'notifications': await SystemNotifications.load(); break;
        }
    },

    // ... reste des méthodes déléguées (inchangé)
};
